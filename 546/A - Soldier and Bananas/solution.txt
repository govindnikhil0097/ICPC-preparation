import sys
import math
def func():
    l = sys.stdin.read().split()
    if not l:
        return
    k = int(l[0])
    n = int(l[1])
    w = int(l[2])
    sum = k * ((w*(w+1))//2)
    reqdAmt = sum- n
    if reqdAmt > 0:
        print(reqdAmt)
    else:
        print(0)
 
if __name__ == "__main__":
    func()
